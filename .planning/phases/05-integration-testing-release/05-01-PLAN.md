4. Parse dist/index.d.mts and verify it contains explicit export signatures (no inferred types)
5. Ensure no "implicit any" errors in generated declarations

Use Node.js child_process or execa to run:
- `pnpm typecheck` (should pass with isolatedDeclarations: true)
- `pnpm build` (should generate .d.ts without errors)

Example pattern:
```typescript
import { execSync } from "node:child_process"
import { existsSync, readFileSync } from "node:fs"

describe("isolatedDeclarations compliance", () => {
  it("tsconfig enforces isolatedDeclarations", () => {
    const tsconfig = JSON.parse(readFileSync("tsconfig.json", "utf-8"))
    expect(tsconfig.compilerOptions.isolatedDeclarations).toBe(true)
  })

  it("build emits .d.ts files successfully", () => {
    execSync("pnpm build", { encoding: "utf-8" })
    expect(existsSync("dist/index.d.mts")).toBe(true)
    expect(existsSync("dist/index.d.cts")).toBe(true)
  })
})
```
